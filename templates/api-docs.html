Username Scan

  -H "Content-Type: application/json" \
  -H "X-API-Key: YOUR_API_KEY" \
  -d '{"username": "john_doe"}'

Python Example
import requests

API  = "http://127.0.0.1:5000/api/v1"
KEY  = "YOUR_API_KEY"
HEAD = {"X-API-Key": KEY, "Content-Type": "application/json"}

# IP Lookup
r = requests.post(f"{API}/ip", json={"ip": "8.8.8.8"}, headers=HEAD)
print(r.json())
